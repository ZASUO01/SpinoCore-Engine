21 inline std::mt19937& GetEngine() noexcept {
22 thread_local std::mt19937 rng = [] {
23 std::random_device rd;
24
25 std::array<std::mt19937::result_type, std::mt19937::state_size> seedData{};
26 for (auto& s : seedData) {
