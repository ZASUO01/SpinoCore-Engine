51 template <std::floating_point T>
52 [[nodiscard]] Vector3<T> GetRotationEuler(const Matrix4<T>& mat) noexcept {
53 const Vector3<T> right = GetRight(mat).Normalized();
54 const Vector3<T> up = GetUp(mat).Normalized();
55 const Vector3<T> back = (-GetForward(mat)).Normalized();
56
57 Vector3<T> euler;
58
59 const T sinPitch = -back.y;
60 euler.x = Operations::Asin(sinPitch);
61
62 if (Operations::Abs(sinPitch) < static_cast<T>(0.9999)) {
63 euler.y = Operations::Atan2(back.x, back.z);
64 euler.z = Operations::Atan2(right.y, up.y);
65 } else {
66 euler.y = Operations::Atan2(-right.z, right.x);
67 euler.z = T{0};
68 }
69
70 return euler;
71 }

179 template <std::floating_point T>
180 [[nodiscard]] constexpr Matrix4<T> CreateFromQuaternion(const Quaternion<T>& q) noexcept {
181 Matrix4<T> mat;
182
183 const T xx = q.x * q.x;
184 const T yy = q.y * q.y;
185 const T zz = q.z * q.z;
186
187 const T xy = q.x * q.y;
188 const T xz = q.x * q.z;
189 const T yz = q.y * q.z;
190
191 const T wx = q.w * q.x;
192 const T wy = q.w * q.y;
193 const T wz = q.w * q.z;
194
195 mat(0, 0) = T{1} - T{2} * (yy + zz);
196 mat(1, 0) = T{2} * (xy + wz);
197 mat(2, 0) = T{2} * (xz - wy);
198
199 mat(0, 1) = T{2} * (xy - wz);
200 mat(1, 1) = T{1} - T{2} * (xx + zz);
201 mat(2, 1) = T{2} * (yz + wx);
202
203 mat(0, 2) = T{2} * (xz + wy);
204 mat(1, 2) = T{2} * (yz - wx);
205 mat(2, 2) = T{1} - T{2} * (xx + yy);
206
207 return mat;
208 }

226 template <std::floating_point T>
227 [[nodiscard]] Matrix4<T> CreateLookAt(const Vector3<T>& eye, const Vector3<T>& target, const Vector3<T>& up) noexcept {
228 Vector3<T> forward = target - eye;
229 const T forwardLengthSq = forward.LengthSq();
230 if (forwardLengthSq < static_cast<T>(1e-8)) return Matrix4<T>();
231 forward /= Operations::Sqrt(forwardLengthSq);
232
233 Vector3<T> right = forward.Cross(up);
234 const T rightLengthSq = right.LengthSq();
235 if (rightLengthSq < static_cast<T>(1e-8)) {
236 const Vector3<T> fallbackUp = Operations::Abs(forward.y) > static_cast<T>(0.99) ? Vector3<T>::UnitZ() : Vector3<T>::UnitY();
237 right = forward.Cross(fallbackUp).Normalized();
238 } else {
239 right /= Operations::Sqrt(rightLengthSq);
240 }
241
242 const Vector3<T> trueUp = right.Cross(forward);
243
244 Matrix4<T> mat;
245
246 mat(0, 0) = right.x;
247 mat(0, 1) = right.y;
248 mat(0, 2) = right.z;
249 mat(0, 3) = -right.Dot(eye);
250
251 mat(1, 0) = trueUp.x;
252 mat(1, 1) = trueUp.y;
253 mat(1, 2) = trueUp.z;
254 mat(1, 3) = -trueUp.Dot(eye);
255
256 mat(2, 0) = -forward.x;
257 mat(2, 1) = -forward.y;
258 mat(2, 2) = -forward.z;
259 mat(2, 3) = forward.Dot(eye);
260
261 return mat;
262 }

271 [[nodiscard]] constexpr Matrix4<T> CreateOrtho(const T width, const T height, const T nearPlane, const T farPlane) noexcept {
272 Matrix4<T> mat;
273
274 if (width <= T{0} || height <= T{0} || nearPlane == farPlane) return mat;
275
276 mat(0, 0) = T{2} / width;
277 mat(1, 1) = T{2} / height;
278 mat(2, 2) = T{-2} / (farPlane - nearPlane);
279 mat(2, 3) = -(farPlane + nearPlane) / (farPlane - nearPlane);
280
281 return mat;
282 }

292 template <std::floating_point T>
293 [[nodiscard]] constexpr Matrix4<T> CreateOrtho(const T left, const T right, const T bottom, const T top,
294 const T nearPlane, const T farPlane) noexcept {
295 Matrix4<T> mat;
296
297 if (left == right || top == bottom || nearPlane == farPlane) return mat;
298
299 mat(0, 0) = T{2} / (right - left);
300 mat(1, 1) = T{2} / (top - bottom);
301 mat(2, 2) = T{-2} / (farPlane - nearPlane);
302
303 mat(0, 3) = -(right + left) / (right - left);
304 mat(1, 3) = -(top + bottom) / (top - bottom);
305 mat(2, 3) = -(farPlane + nearPlane) / (farPlane - nearPlane);
306
307 return mat;

317 template <std::floating_point T>
318 [[nodiscard]] Matrix4<T> CreatePerspectiveFOV(const T fovYRadians, const T width, const T height,
319 const T nearPlane, const T farPlane) noexcept {
320 Matrix4<T> mat;
321
322 if (width <= T{0} || height <= T{0} || nearPlane == farPlane) return mat;
323
324 const T yScale = T{1} / Operations::Tan(fovYRadians * T{0.5});
325 const T xScale = yScale * height / width;
326
327 mat(0, 0) = xScale;
328 mat(1, 1) = yScale;
329 mat(2, 2) = (farPlane + nearPlane) / (nearPlane - farPlane);
330 mat(2, 3) = T{2} * farPlane * nearPlane / (nearPlane - farPlane);
331 mat(3, 2) = T{-1};
332 mat(3, 3) = T{0};
333
334 return mat;
335 }
