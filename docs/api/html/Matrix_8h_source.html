85 [[nodiscard]] constexpr Matrix3 GetInverse(bool* outSuccess = nullptr) const noexcept requires std::floating_point<T> {
86 Matrix3 inv;
87
88 const T cofactor00 = data[4] * data[8] - data[7] * data[5];
89 const T cofactor10 = data[7] * data[2] - data[1] * data[8];
90 const T cofactor20 = data[1] * data[5] - data[4] * data[2];
91
92 const T cofactor01 = data[6] * data[5] - data[3] * data[8];
93 const T cofactor11 = data[0] * data[8] - data[6] * data[2];
94 const T cofactor21 = data[3] * data[2] - data[0] * data[5];
95
96 const T cofactor02 = data[3] * data[7] - data[6] * data[4];
97 const T cofactor12 = data[6] * data[1] - data[0] * data[7];
98 const T cofactor22 = data[0] * data[4] - data[3] * data[1];
99
100 const T det = data[0] * cofactor00 + data[3] * cofactor10 + data[6] * cofactor20;
101
102 if (det == T{0}) {
103 if (outSuccess) *outSuccess = false;
104 return Matrix3();
105 }
106
107 const T invDet = T{1} / det;
108
109 inv.data[0] = cofactor00 * invDet;
110 inv.data[1] = cofactor10 * invDet;
111 inv.data[2] = cofactor20 * invDet;
112
113 inv.data[3] = cofactor01 * invDet;
114 inv.data[4] = cofactor11 * invDet;
115 inv.data[5] = cofactor21 * invDet;
116
117 inv.data[6] = cofactor02 * invDet;
118 inv.data[7] = cofactor12 * invDet;
119 inv.data[8] = cofactor22 * invDet;
120
121 if (outSuccess) *outSuccess = true;
122 return inv;
123 }

241 [[nodiscard]] constexpr Matrix4 GetInverse(bool* outSuccess = nullptr) const noexcept requires std::floating_point<T> {
242 Matrix4 inv;
243
244 const T subFactor00 = data[10] * data[15] - data[14] * data[11];
245 const T subFactor01 = data[6] * data[15] - data[14] * data[7];
246 const T subFactor02 = data[6] * data[11] - data[10] * data[7];
247 const T subFactor03 = data[2] * data[15] - data[14] * data[3];
248 const T subFactor04 = data[2] * data[11] - data[10] * data[3];
249 const T subFactor05 = data[2] * data[7] - data[6] * data[3];
250 const T subFactor06 = data[8] * data[13] - data[12] * data[9];
251 const T subFactor07 = data[4] * data[13] - data[12] * data[5];
252 const T subFactor08 = data[4] * data[9] - data[8] * data[5];
253 const T subFactor09 = data[0] * data[13] - data[12] * data[1];
254 const T subFactor10 = data[0] * data[9] - data[8] * data[1];
255 const T subFactor11 = data[0] * data[5] - data[4] * data[1];
256
257 inv.data[0] = data[5] * subFactor00 - data[9] * subFactor01 + data[13] * subFactor02;
258 inv.data[1] = -(data[1] * subFactor00 - data[9] * subFactor03 + data[13] * subFactor04);
259 inv.data[2] = data[1] * subFactor01 - data[5] * subFactor03 + data[13] * subFactor05;
260 inv.data[3] = -(data[1] * subFactor02 - data[5] * subFactor04 + data[9] * subFactor05);
261
262 inv.data[4] = -(data[4] * subFactor00 - data[8] * subFactor01 + data[12] * subFactor02);
263 inv.data[5] = data[0] * subFactor00 - data[8] * subFactor03 + data[12] * subFactor04;
264 inv.data[6] = -(data[0] * subFactor01 - data[4] * subFactor03 + data[12] * subFactor05);
265 inv.data[7] = data[0] * subFactor02 - data[4] * subFactor04 + data[8] * subFactor05;
266
267 inv.data[8] = data[7] * subFactor06 - data[11] * subFactor07 + data[15] * subFactor08;
268 inv.data[9] = -(data[3] * subFactor06 - data[11] * subFactor09 + data[15] * subFactor10);
269 inv.data[10] = data[3] * subFactor07 - data[7] * subFactor09 + data[15] * subFactor11;
270 inv.data[11] = -(data[3] * subFactor08 - data[7] * subFactor10 + data[11] * subFactor11);
271
272 inv.data[12] = -(data[6] * subFactor06 - data[10] * subFactor07 + data[14] * subFactor08);
273 inv.data[13] = data[2] * subFactor06 - data[10] * subFactor09 + data[14] * subFactor10;
274 inv.data[14] = -(data[2] * subFactor07 - data[6] * subFactor09 + data[14] * subFactor11);
275 inv.data[15] = data[2] * subFactor08 - data[6] * subFactor10 + data[10] * subFactor11;
276
277 const T det = data[0] * inv.data[0] + data[1] * inv.data[4] + data[2] * inv.data[8] + data[3] * inv.data[12];
278
279 if (det == T{0}) {
280 if (outSuccess) *outSuccess = false;
281 return Matrix4();
282 }
283
284 const T invDet = T{1} / det;
285
286 for (auto & i : inv.data) {
287 i *= invDet;
288 }
289
290 if (outSuccess) *outSuccess = true;
291 return inv;
292 }

344 [[nodiscard]] constexpr Vector3<T> TransformPoint(const Vector3<T>& v) const noexcept requires std::floating_point<T> {
345 T x = v.x * operator()(0, 0) + v.y * operator()(0, 1) + v.z * operator()(0, 2) + operator()(0, 3);
346 T y = v.x * operator()(1, 0) + v.y * operator()(1, 1) + v.z * operator()(1, 2) + operator()(1, 3);
347 T z = v.x * operator()(2, 0) + v.y * operator()(2, 1) + v.z * operator()(2, 2) + operator()(2, 3);
348 T w = v.x * operator()(3, 0) + v.y * operator()(3, 1) + v.z * operator()(3, 2) + operator()(3, 3);
349
350 if (w != T{1} && !Operations::NearlyZero(w)) {
351 const T invW = T{1} / w;
352 x *= invW; y *= invW; z *= invW;
353 }
354
355 return Vector3<T>{x, y, z};
