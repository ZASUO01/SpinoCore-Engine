129 [[nodiscard]] static Quaternion Lerp(const Quaternion& start, const Quaternion& end, T t) noexcept {
130 T cosTheta = start.Dot(end);
131 Quaternion result;
132
133 if (cosTheta < T{0}) {
134 result.x = start.x + (-end.x - start.x) * t;
135 result.y = start.y + (-end.y - start.y) * t;
136 result.z = start.z + (-end.z - start.z) * t;
137 result.w = start.w + (-end.w - start.w) * t;
138 } else {
139 result.x = start.x + (end.x - start.x) * t;
140 result.y = start.y + (end.y - start.y) * t;
141 result.z = start.z + (end.z - start.z) * t;
142 result.w = start.w + (end.w - start.w) * t;
143 }
144
145 return result.Normalized();
146 }

152 [[nodiscard]] static Quaternion Slerp(Quaternion start, Quaternion end, T t) noexcept {
153 T cosTheta = start.Dot(end);
154 Quaternion endTarget = end;
155
156 if (cosTheta < T{0}) {
157 cosTheta = -cosTheta;
158 endTarget.x = -end.x;
159 endTarget.y = -end.y;
160 endTarget.z = -end.z;
161 endTarget.w = -end.w;
162 }
163
164 if (cosTheta > static_cast<T>(0.99999)) {
165 return Lerp(start, endTarget, t);
166 }
167
168 const T halfTheta = Operations::Acos(cosTheta);
169 const T sinHalfTheta = Operations::Sqrt<T>(T{1} - cosTheta * cosTheta);
170
171 const T invSinHalfTheta = T{1} / sinHalfTheta;
172 const T weightStart = Operations::Sin((T{1} - t) * halfTheta) * invSinHalfTheta;
173 const T weightEnd = Operations::Sin(t * halfTheta) * invSinHalfTheta;
174
175 return Quaternion(
176 start.x * weightStart + endTarget.x * weightEnd,
177 start.y * weightStart + endTarget.y * weightEnd,
178 start.z * weightStart + endTarget.z * weightEnd,
179 start.w * weightStart + endTarget.w * weightEnd
180 );
181 }

188 [[nodiscard]] static Quaternion FromEuler(T pitch, T yaw, T roll) noexcept {
189 const T halfPitch = pitch * T{0.5};
190 const T halfYaw = yaw * T{0.5};
191 const T halfRoll = roll * T{0.5};
192
193 const T cx = Operations::Cos(halfPitch);
194 const T sx = Operations::Sin(halfPitch);
195 const T cy = Operations::Cos(halfYaw);
196 const T sy = Operations::Sin(halfYaw);
197 const T cz = Operations::Cos(halfRoll);
198 const T sz = Operations::Sin(halfRoll);
199
200 return Quaternion(
201 cy * sx * cz + sy * cx * sz,
202 sy * cx * cz - cy * sx * sz,
203 cy * cx * sz - sy * sx * cz,
204 cy * cx * cz + sy * sx * sz
205 );
206 }

213 [[nodiscard]] static Quaternion FromAxes(const Vector3<T>& right, const Vector3<T>& up, const Vector3<T>& back) noexcept {
214 const T trace = right.x + up.y + back.z;
215
216 if (trace > T{0}) {
217 const T s = Operations::Sqrt(trace + T{1}) * T{2};
218 return Quaternion((up.z - back.y) / s, (back.x - right.z) / s, (right.y - up.x) / s, s / T{4});
219 }
220 if (right.x > up.y && right.x > back.z) {
221 const T s = Operations::Sqrt(T{1} + right.x - up.y - back.z) * T{2};
222 return Quaternion(s / T{4}, (up.x + right.y) / s, (back.x + right.z) / s, (up.z - back.y) / s);
223 }
224 if (up.y > back.z) {
225 const T s = Operations::Sqrt(T{1} + up.y - right.x - back.z) * T{2};
226 return Quaternion((up.x + right.y) / s, s / T{4}, (back.y + up.z) / s, (back.x - right.z) / s);
227 }
228 const T s = Operations::Sqrt(T{1} + back.z - right.x - up.y) * T{2};
229 return Quaternion((back.x + right.z) / s, (back.y + up.z) / s, s / T{4}, (right.y - up.x) / s);
230 }

236 [[nodiscard]] static Quaternion LookRotation(const Vector3<T>& forward, const Vector3<T>& up = Vector3<T>::UnitY()) noexcept {
237 const T forwardLengthSq = forward.LengthSq();
238 if (forwardLengthSq <= T{0}) return Identity();
239
240 const Vector3<T> back = -forward / Operations::Sqrt(forwardLengthSq);
241 Vector3<T> right = up.Cross(back);
242 if (right.LengthSq() <= static_cast<T>(1e-8)) {
243 const Vector3<T> fallbackUp = Operations::Abs(back.y) > static_cast<T>(0.99) ? Vector3<T>::UnitZ() : Vector3<T>::UnitY();
244 right = fallbackUp.Cross(back);
245 }
246 right.Normalize();
247
248 return FromAxes(right, back.Cross(right), back);
249 }

253 [[nodiscard]] constexpr Vector3<T> Rotate(const Vector3<T>& v) const noexcept {
254 const Vector3<T> axis{x, y, z};
255 const Vector3<T> t = axis.Cross(v) * T{2};
256 return v + t * w + axis.Cross(t);
257 }

273 [[nodiscard]] static constexpr bool NearlyEqual(const Quaternion& a, const Quaternion& b,
274 T epsilon = Constants::Tolerance<T>) noexcept {
275 const auto sameComponents = [epsilon](const Quaternion& lhs, const Quaternion& rhs) {
276 return Operations::NearlyEqual(lhs.x, rhs.x, epsilon) && Operations::NearlyEqual(lhs.y, rhs.y, epsilon) &&
277 Operations::NearlyEqual(lhs.z, rhs.z, epsilon) && Operations::NearlyEqual(lhs.w, rhs.w, epsilon);
278 };
279 return sameComponents(a, b) || sameComponents(a, Quaternion(-b.x, -b.y, -b.z, -b.w));
280 }
